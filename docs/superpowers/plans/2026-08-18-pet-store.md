# PetNest 宠物商店实现计划

> **面向 AI 代理的工作者：** 必需子技能：使用 superpowers:subagent-driven-development（推荐）或 superpowers:executing-plans 逐任务实现此计划。步骤使用复选框（`- [ ]`）语法来跟踪进度。

**目标：** 在 PetNest 的“宠物与动作”窗口中增加官方精选宠物商店，并通过现有私有资源仓库与 Worker 安全提供浏览、轻量预览、领养和摘要驱动的更新。

**架构：** 私有 `petnest-resources` 仓库保存生成的商店目录、封面、预览精灵图和完整宠物 ZIP；Worker 用独立白名单公开目录与带 SHA-256 的不可变文件。桌面客户端把目录解析、缓存下载、领养记录、安装编排和 Qt 页面分层，最终复用现有 `import_pet_package`、备份和原子替换能力；普通导入继续自动切换，商店领养只刷新宠物库。

**技术栈：** Python 3.12、PySide6、Pillow、pytest/pytest-qt、Cloudflare Workers JavaScript、Node.js test runner、GitHub Contents API。

```python
class StoreCatalogBuildError(ValueError):
    pass


def build_catalog(root: Path, *, now: datetime) -> dict[str, object]:
    store_root = root.resolve() / "store"
    previous = _load_previous_catalog(store_root / "catalog.json")
    pets = [_build_pet(path, root.resolve(), previous, now) for path in _pet_directories(store_root)]
    ids = [str(item["id"]) for item in pets]
    if len(ids) != len(set(ids)):
        raise StoreCatalogBuildError("商店宠物 ID 不能重复")
    featured = [str(item["id"]) for item in pets if item.pop("featured", False)]
    if pets and len(featured) != 1:
        raise StoreCatalogBuildError("商店必须且只能配置一个推荐宠物")
    return {
        "schema_version": 1,
        "generated_at": _utc_text(now),
        "featured_pet_id": featured[0] if featured else None,
        "pets": pets,
    }
```

每个条目必须包含 `id/name/author/summary/tags/updated_at/action_count/capabilities/cover/idle_preview/package`。`_build_pet` 读取 ZIP 根目录的 `pet.json`，验证 ID 与商品目录名一致；`idle_preview` 从横向 PNG 宽度和 ZIP 内 idle 帧数推导单帧宽高，并从 `frame_durations_ms` 或 FPS 生成时间线。文件记录使用二进制大小和 SHA-256。

命令行接口固定为 `python tools/build_store_catalog.py --root <repo> --output store/catalog.json --now <ISO-8601>`；`--now` 省略时使用当前 UTC，测试使用显式时间保证可重复。

```python
class PublishOptions:
    summary: str
    tags: tuple[str, ...]
    featured: bool = False
    preview_frame_limit: int = 8


def publish_pet(
    source: Path,
    repository_root: Path,
    *,
    summary: str,
    tags: tuple[str, ...],
    featured: bool = False,
) -> Path:
    options = PublishOptions(summary.strip(), tuple(dict.fromkeys(tags)), featured)
    package = _inspect_source(source.resolve(), options)
    destination = repository_root.resolve() / "store" / "pets" / package.identifier

    _write_cover(destination, package)
```

```python
@dataclass(frozen=True, slots=True)
class PetStoreFile:
    path: str
    size: int
    sha256: str

    @property
    def relative_path(self) -> PurePosixPath:
        return PurePosixPath(self.path)


@dataclass(frozen=True, slots=True)
class PetStorePreview:
    file: PetStoreFile
    frame_width: int
    frame_height: int
    frame_count: int
    frame_durations_ms: tuple[int, ...]


@dataclass(frozen=True, slots=True)
class PetStoreItem:
    identifier: str
    name: str
    author: str
    summary: str
    tags: tuple[str, ...]
    updated_at: datetime
    action_count: int
    capabilities: tuple[str, ...]
    cover: PetStoreFile
    idle_preview: PetStorePreview
    package: PetStoreFile


@dataclass(frozen=True, slots=True)
class PetStoreCatalog:
    generated_at: datetime
    featured_pet_id: str
    pets: tuple[PetStoreItem, ...]

    @classmethod
    def from_bytes(cls, payload: bytes) -> "PetStoreCatalog":
        if len(payload) > MAX_CATALOG_BYTES:
            raise PetStoreCatalogError("商店目录超过大小限制")
        try:
            raw = json.loads(payload.decode("utf-8"))
        except (UnicodeDecodeError, json.JSONDecodeError) as error:
            raise PetStoreCatalogError(f"商店目录不是有效 JSON：{error}") from error
        if not isinstance(raw, Mapping):
            raise PetStoreCatalogError("商店目录顶层必须是对象")
        return cls.from_dict(raw)

    def pet(self, identifier: str) -> PetStoreItem | None:
        return next((item for item in self.pets if item.identifier == identifier), None)
```

```python
class PetStoreStatus(StrEnum):
    NOT_ADOPTED = "not_adopted"
    LOCAL_EXISTING = "local_existing"
    ADOPTED = "adopted"
    UPDATE_AVAILABLE = "update_available"


@dataclass(frozen=True, slots=True)
class PetStoreReceipt:
    pet_id: str
    package_sha256: str
    installed_at: datetime
    catalog_updated_at: datetime


class PetStoreStateStore:
    def status_for(self, item: PetStoreItem, pets_root: Path) -> PetStoreStatus:
        exists = (Path(pets_root) / item.identifier).is_dir()
        receipt = self.receipt(item.identifier)
        if not exists:
            if receipt is not None:
                self.forget(item.identifier)
            return PetStoreStatus.NOT_ADOPTED
        if receipt is None:
            return PetStoreStatus.LOCAL_EXISTING
        if receipt.package_sha256 == item.package.sha256:
            return PetStoreStatus.ADOPTED
        return PetStoreStatus.UPDATE_AVAILABLE
```

```python
def rollback_pet_import(result: PetImportResult, pets_root: Path) -> None:
    root = Path(pets_root).expanduser().resolve()
```
